const W = 180, H = 66;
const PADX = 30, PADY = 26;

function separate(list, fixedId) {
  const out = list.map(n => ({ ...n }));
  const needX = W + PADX, needY = H + PADY;
  for (let pass = 0; pass < 10; pass++) {
    let any = false;
    for (let i = 0; i < out.length; i++) for (let j = i + 1; j < out.length; j++) {
      const a = out[i], b = out[j];
      const dx = b.x - a.x, dy = b.y - a.y;
      const ox = needX - Math.abs(dx), oy = needY - Math.abs(dy);
      if (ox > 0 && oy > 0) { any = true;
        if (ox <= oy) { const dir = dx < 0 ? -1 : 1;
          if (a.id === fixedId) b.x += dir * ox; else if (b.id === fixedId) a.x -= dir * ox; else { a.x -= dir * ox / 2; b.x += dir * ox / 2; }
        } else { const dir = dy < 0 ? -1 : 1;
          if (a.id === fixedId) b.y += dir * oy; else if (b.id === fixedId) a.y -= dir * oy; else { a.y -= dir * oy / 2; b.y += dir * oy / 2; }
        }
      }
    }
    if (!any) break;
  }
  return out;
}

          {regions.map(r => (<g key={r.t}>
            <rect x={r.x} y={r.y} width={r.w} height={r.h} rx="18" className="tw-region" />
            <text x={r.x+16} y={r.y+23} className="tw-region-lbl">{TYPES[r.t].label.toUpperCase()}</text>
          </g>))}

          {links.map((l,i) => { const a=nodeById[l.from], b=nodeById[l.to]; if(!a||!b) return null;
            const dim = !(isVisible(a)&&isVisible(b)); const hot = selectedId && (l.from===selectedId||l.to===selectedId);
            return <path key={"ml"+i} d={edgePath(a,b)} className={"tw-link"+(dim?" dim":"")+(hot?" hot":"")} markerEnd="url(#arrow-teal)" onClick={()=>delLink(l.from,l.to)} />; })}

          {flagEdges.map((e,i) => { const a=nodeById[e.from], b=nodeById[e.to]; if(!a||!b) return null;
            const dim = !(isVisible(a)&&isVisible(b)); const hot = selectedId && (e.from===selectedId||e.to===selectedId);
            const mx=(a.x+b.x)/2, my=(a.y+b.y)/2;
            return <g key={"fe"+i} className={dim?"dim":""}>
              <path d={edgePath(a,b)} className={"tw-flagedge"+(hot?" hot":"")} markerEnd="url(#arrow-gold)" />
              <text x={mx} y={my-4} className="tw-flagedge-lbl">{e.flag}</text></g>; })}

          {Object.entries(danglingByNode).map(([id,flags]) => { const nd=nodeById[id]; if(!nd||!isVisible(nd)) return null;
            return flags.map((f,j) => { const sx=nd.x+W/2, sy=nd.y-8+j*15, ex=sx+70;
              return <g key={id+f} className="tw-dangle">
                <path d={`M ${sx} ${sy} L ${ex} ${sy}`} className="tw-dangle-line" />
                <circle cx={ex} cy={sy} r="3" className="tw-dangle-dot" />
                <text x={ex+6} y={sy+3} className="tw-dangle-lbl">{f}</text></g>; }); })}

          {encounters.map(e => { const fc=FACTIONS[e.faction]?.color||"#888"; const vis=isVisible(e);
            const sel=e.id===selectedId, cf=connectFrom===e.id;
            return (<g key={e.id} transform={`translate(${e.x-W/2} ${e.y-H/2})`} className={"tw-node"+(vis?"":" dim")+(sel?" sel":"")+(cf?" cf":"")} onPointerDown={ev=>onNodeDown(ev,e.id)} style={{cursor:mode==="connect"?"crosshair":"grab"}}>
              <rect width={W} height={H} rx="9" fill="#211f1b" />
              <rect width={W} height={H} rx="9" className="tw-node-bg" style={{ fill:fc+(sel?"5e":"40"), stroke:fc, strokeWidth:sel?2.5:1.6 }} />
              <rect width="6" height={H} rx="3" style={{ fill:fc }} />
              <circle cx={W-14} cy={14} r="4.5" style={{ fill:STATUSES[e.status].color, stroke:"#1b1a17", strokeWidth:1 }} />
              <text x={16} y={24} className="tw-node-title">{e.title.length>21?e.title.slice(0,20)+"…":e.title}</text>
              <text x={16} y={41} className="tw-node-meta">{TYPES[e.type].label} · {e.scope==="gated"?"gated":"universal"}</text>
              <text x={16} y={56} className="tw-node-flags">{e.out.length?("sets "+e.out.length):""}{e.in.length?("  needs "+e.in.length):""}{(!e.out.length&&!e.in.length)?"no flags":""}</text>
            </g>); })}
        </g>
      </svg>

      {selected && (
        <aside className="tw-drawer" onPointerDown={e=>e.stopPropagation()}>
          <div className="tw-dr-head">
            <input className="tw-dr-title" value={selected.title} onChange={e=>patch(selected.id,{title:e.target.value})} />
            <button className="tw-dr-x" onClick={()=>setSelectedId(null)}>✕</button>
          </div>
          <div className="tw-dr-row3">
            <label>Type<select value={selected.type} onChange={e=>patch(selected.id,{type:e.target.value})}>{Object.entries(TYPES).map(([k,v])=><option key={k} value={k}>{v.label}</option>)}</select></label>
            <label>Faction<select value={selected.faction} onChange={e=>patch(selected.id,{faction:e.target.value})}>{Object.entries(FACTIONS).map(([k,v])=><option key={k} value={k}>{v.label}</option>)}</select></label>
            <label>Status<select value={selected.status} onChange={e=>patch(selected.id,{status:e.target.value})}>{Object.entries(STATUSES).map(([k,v])=><option key={k} value={k}>{v.label}</option>)}</select></label>
          </div>
          <div className="tw-dr-row2">
            <label>Scope<select value={selected.scope} onChange={e=>patch(selected.id,{scope:e.target.value})}><option value="universal">universal</option><option value="gated">faction-gated</option></select></label>
            <label>id<input className="tw-mono" value={selected.id} onChange={e=>patch(selected.id,{id:e.target.value})} /></label>
          </div>
          <label className="tw-dr-field">Hook <span className="tw-dr-hintlbl">what the player sees</span><textarea rows={2} value={selected.hook} onChange={e=>patch(selected.id,{hook:e.target.value})} /></label>
          <label className="tw-dr-field">Teaches <span className="tw-dr-hintlbl">the world-fact this exposes</span><textarea rows={2} value={selected.teaches} onChange={e=>patch(selected.id,{teaches:e.target.value})} /></label>
          <div className="tw-dr-field">
            <div className="tw-dr-choicehead">Choices <button className="tw-mini" onClick={()=>addChoice(selected.id)}>+ add</button></div>
            {selected.choices.map((c,i) => (<div key={i} className="tw-choice">
              <input className="tw-ch-label" placeholder="button text" value={c.label} onChange={e=>patchChoice(selected.id,i,{label:e.target.value})} />
              <input className="tw-ch-fx" placeholder="effect in plain words" value={c.fx} onChange={e=>patchChoice(selected.id,i,{fx:e.target.value})} />
              <button className="tw-ch-x" onClick={()=>delChoice(selected.id,i)}>✕</button></div>))}
          </div>
          <div className="tw-dr-row2">
            <label className="tw-dr-field">Sets flags <span className="tw-dr-hintlbl">comma-sep</span><input className="tw-mono" value={selected.out.join(", ")} onChange={e=>patch(selected.id,{out:parseFlags(e.target.value)})} /></label>
            <label className="tw-dr-field">Needs flags <span className="tw-dr-hintlbl">comma-sep</span><input className="tw-mono" value={selected.in.join(", ")} onChange={e=>patch(selected.id,{in:parseFlags(e.target.value)})} /></label>
          </div>
          <label className="tw-dr-field">Notes for Claude<textarea rows={2} value={selected.notes} onChange={e=>patch(selected.id,{notes:e.target.value})} /></label>
          <div className="tw-dr-foot">
            <button className="tw-btn ghost" onClick={()=>{ setMode("connect"); setConnectFrom(selected.id); }}>Link from here</button>
            <button className="tw-btn danger" onClick={()=>delNode(selected.id)}>Delete</button>
          </div>
        </aside>
      )}

      {modal && (
        <div className="tw-modal-wrap" onPointerDown={()=>setModal(null)}>
          <div className="tw-modal" onPointerDown={e=>e.stopPropagation()}>
            <div className="tw-modal-h">{modal==="export"?"Export — paste this to Claude to mechanize":"Import — paste a board JSON"}</div>
            <textarea className="tw-io" value={ioText} onChange={e=>setIoText(e.target.value)} spellCheck={false} />
            <div className="tw-modal-foot">
              {modal==="export"
                ? <><button className="tw-btn" onClick={copyExport}>Copy</button><button className="tw-btn ghost" onClick={()=>setModal(null)}>Close</button></>
                : <><button className="tw-btn primary" onClick={doImport}>Load board</button><button className="tw-btn ghost" onClick={()=>setModal(null)}>Cancel</button></>}
            </div>
          </div>
        </div>
      )}
    </div>
  );
}
